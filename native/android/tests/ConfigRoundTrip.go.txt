package main

import (
    "fmt"
    "os"
    "github.com/psportal/psportal/native/engine"
)
func main() {
    bytes, err := os.ReadFile(os.Args[1]); if err != nil { panic(err) }
    config, err := engine.ParseConfig(string(bytes)); if err != nil { panic(err) }
    switch config.SelectionMode {
    case "fixed":
        if config.PreferredNode != "香港 A" || config.AllowedClients[0] != "192.168.1.30" { panic("fixed roundtrip mismatch") }
    case "auto":
        if len(config.AllowedClients) != 253 { panic("subnet roundtrip mismatch") }
        for _, ip := range config.AllowedClients { if ip == config.ListenHost || ip == "192.168.1.0" || ip == "192.168.1.255" { panic("unsafe subnet client") } }
    case "fastest":
        if config.SwitchThresholdMS != 75 || config.PreferredNode != "" { panic("fastest roundtrip mismatch") }
    default: panic("unexpected mode")
    }
    fmt.Println("PASS: Android-generated", config.SelectionMode, "config accepted by real Go engine.ParseConfig")
}
